// ===== Inför kvällen: gänget (namn och vems tur), kvällens lista och timer. Allt sparas bara i telefonen. =====
const GANG = (() => { const v = lasLS('dryckeslexikon-ganget', {}); return {namn: Array.isArray(v.namn) ? v.namn.filter(x => typeof x === 'string' && x.trim()).slice(0, 30) : [], tur: Math.max(0, v.tur|0)}; })();
const sparaGang = () => skrivLS('dryckeslexikon-ganget', GANG);
const gangN = () => GANG.namn.length;
const turNamn = (k = 0) => gangN() ? GANG.namn[(((GANG.tur + k) % gangN()) + gangN()) % gangN()] : null;
const flyttaTur = k => { if (gangN()){ GANG.tur = (((GANG.tur + k) % gangN()) + gangN()) % gangN(); sparaGang(); } };
let KV = lasLS('dryckeslexikon-kvallen', []).filter(id => LEKAR.some(l => l.id === id));
const sparaKV = () => skrivLS('dryckeslexikon-kvallen', KV);
let KVAKTIV = false;

function inforKvallen(){
  return `<nav class="ikvall" aria-label="${T('Inför kvällen')}">
    <a class="kvl" href="#kvallen">${ikon('bok')}<span>${T('Kvällens lista')}<small>${KV.length ? T('{n} lekar valda', {n: KV.length}) : T('Välj lekar inför festen')}</small></span></a>
    <a class="gl" href="#ganget">${ikon('folk')}<span>${T('Gänget')}<small>${gangN() ? esc(GANG.namn.join(', ')) : T('Skriv in vilka som är med')}</small></span></a>
  </nav>`;
}

// Vems tur? (leksidan och gänget)
function turHtml(){
  return gangN() ? `<div class="turruta"><div><small>${T('Vems tur?')}</small><b id="turnamn">${esc(turNamn())}</b></div>
      <button type="button" class="liten" data-tur="slumpa">${T('🎲 Vem börjar?')}</button><button type="button" class="liten" data-tur="nasta" aria-label="${T('Nästa i tur')}">${T('Nästa →')}</button></div>`
    : `<a class="turtom" href="#ganget">${T('👥 Skriv in gänget så slumpar appen vem som börjar')}</a>`;
}
let TURANIM = null;
function slumpaTur(){
  const n = gangN(); if (!n) return;
  const mal = Math.floor(Math.random() * n), el = () => document.getElementById('turnamn');
  clearInterval(TURANIM);
  const klart = () => { GANG.tur = mal; sparaGang(); if (location.hash.slice(1) === 'ganget') gangSida(); const e = el(); if (e){ e.textContent = GANG.namn[mal]; e.classList.remove('pop'); void e.offsetWidth; e.classList.add('pop'); } try { navigator.vibrate && navigator.vibrate(60); } catch(x) {} };
  if (n === 1 || matchMedia('(prefers-reduced-motion: reduce)').matches) return klart();
  let steg = 0, i = GANG.tur;
  TURANIM = setInterval(() => { i = (i + 1 + Math.floor(Math.random() * (n - 1))) % n; const e = el(); if (e) e.textContent = GANG.namn[i]; if (++steg >= 14){ clearInterval(TURANIM); klart(); } }, 75);
}

function gangSida(){
  const n = gangN(), tur = n ? GANG.tur % n : 0;
  $('#vy').innerHTML = `<section class="verktyg gang">
    <h1>${T('Gänget')}</h1>
    <p>${T('Skriv in alla som är med ikväll. Då kan appen slumpa vem som börjar, visa vems tur det är och använda namnen i Impostor, Körkortet och spellägena. Namnen sparas bara i den här telefonen.')}</p>
    <form class="egnarad" id="gangform"><input id="gangin" maxlength="24" placeholder="${esc(T('Namn'))}" aria-label="${esc(T('Namn'))}" autocomplete="off" enterkeyhint="done"><button type="submit">${T('Lägg till')}</button></form>
    ${n ? `<ol class="ganglista">${GANG.namn.map((x, i) => `<li class="${i === tur ? 'tur' : ''}"><span>${esc(x)}</span>${i === tur ? `<em>${T('Tur')}</em>` : ''}<button type="button" data-gtabort="${i}" aria-label="${esc(T('Ta bort {n}', {n: x}))}">✕</button></li>`).join('')}</ol>
      <div class="vemborjar"><small>${T('Vems tur?')}</small><b id="turnamn">${esc(turNamn())}</b></div>
      <button type="button" class="spela" data-tur="slumpa">${T('🎲 Vem börjar?')}</button>
      <div class="knapprad" style="margin-top:12px"><button type="button" class="liten" data-tur="nasta">${T('Nästa i tur')}</button><button type="button" class="liten" data-gang="blanda">${T('Blanda ordningen')}</button><button type="button" class="liten" data-gang="rensa">${T('Töm gänget')}</button></div>`
    : `<p class="tom" style="padding:24px 0">${T('Inga namn än. Skriv ett namn i taget.')}</p>`}
  </section>`;
  $('#gangform').addEventListener('submit', e => { e.preventDefault(); const t = $('#gangin').value.trim().slice(0, 24); if (!t || gangN() >= 30) return;
    GANG.namn.push(t); sparaGang(); gangSida(); $('#gangin').focus(); });
}

function kvBar(l){
  if (!KVAKTIV || !KV.includes(l.id)) { document.body.classList.remove('medkv'); return ''; }
  document.body.classList.add('medkv');
  const i = KV.indexOf(l.id), fo = KV[i - 1], ne = KV[i + 1], nl = ne && LEKAR.find(x => x.id === ne);
  const pil = (id, t, lbl) => id ? `<a class="kvpil" href="#lek-${encodeURIComponent(id)}" aria-label="${lbl}">${t}</a>` : `<a class="kvpil" aria-disabled="true">${t}</a>`;
  return `<div class="kvplats"></div><nav class="kvbar" aria-label="${T('Kvällens lista')}">${pil(fo, '←', T('Förra leken'))}
    <a class="kvmitt" href="#kvallen"><b>${T('{i} av {n}', {i: i + 1, n: KV.length})}</b><small>${nl ? T('Nästa: {x}', {x: esc(F(nl,'namn'))}) : T('Sista leken ikväll')}</small></a>${pil(ne, '→', T('Nästa lek'))}</nav>`;
}

// Timer på lekar som har en tidsgräns i reglerna (t.ex. "1 minut", "30 sekunder").
const TM = {lek:null, std:60, sek:60, kvar:60, slut:0, int:null, ac:null, wl:null};
function timerFor(l){
  const m = [l.kort || '', ...(l.regler || []), ...(l.varianter || [])].join(' ').match(/(\d+)(?:\s*[–-]\s*\d+)?\s*(sekund|minut)/i);
  if (!m) return null; const s = +m[1] * (/minut/i.test(m[2]) ? 60 : 1);
  return s >= 5 && s <= 600 ? s : null;
}
const mmss = s => `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;
const tidText = s => s < 60 ? T('{n} sek', {n: s}) : s % 60 ? mmss(s) : T('{n} min', {n: s / 60});
function timerHtml(){
  const kor = !!TM.int, slut = TM.kvar === 0;
  return `<div class="timer ${kor ? 'kor' : ''} ${slut ? 'slut' : ''}" id="timer"><small>${T('Timer')}</small>
    <div class="tid" role="timer">${slut ? T('Tiden är ute!') : mmss(TM.kvar)}</div>
    <div class="chips">${[...new Set([TM.std, 30, 60, 120])].sort((a, b) => a - b).map(s => `<button type="button" class="chip" data-tsek="${s}" aria-pressed="${TM.sek === s}">${tidText(s)}</button>`).join('')}</div>
    <div class="knapprad"><button type="button" class="liten tstart ${kor ? 'paus' : ''}" data-timer="start">${kor ? T('Paus') : TM.kvar === TM.sek || slut ? T('▶ Starta') : T('▶ Fortsätt')}</button><button type="button" class="liten" data-timer="noll">${T('Börja om')}</button></div></div>`;
}
const ritaTimer = () => { const el = document.getElementById('timer'); if (el) el.outerHTML = timerHtml(); };
function timerStopp(){ clearInterval(TM.int); TM.int = null; try { TM.wl && TM.wl.release(); } catch(e) {} TM.wl = null; }
async function timerVaken(){ try { if ('wakeLock' in navigator && !TM.wl){ TM.wl = await navigator.wakeLock.request('screen'); TM.wl.addEventListener('release', () => TM.wl = null); } } catch(e) {} }
function timerTick(){
  const k = Math.max(0, Math.ceil((TM.slut - Date.now()) / 1000));
  if (k !== TM.kvar){ TM.kvar = k; const el = document.querySelector('#timer .tid'); if (el && k) el.textContent = mmss(k); }
  if (k === 0){ timerStopp(); timerLarm(); ritaTimer(); }
}
function timerStart(){
  try { TM.ac = TM.ac || new (window.AudioContext || window.webkitAudioContext)(); TM.ac.resume && TM.ac.resume(); } catch(e) {}
  if (TM.kvar === 0) TM.kvar = TM.sek;
  TM.slut = Date.now() + TM.kvar * 1000; clearInterval(TM.int); TM.int = setInterval(timerTick, 200); timerVaken(); ritaTimer();
}
function timerLarm(){
  try { navigator.vibrate && navigator.vibrate([400, 150, 400, 150, 800]); } catch(e) {}
  try { const ac = TM.ac; if (!ac) return; const t0 = ac.currentTime;
    [0, .35, .7].forEach(d => { const o = ac.createOscillator(), g = ac.createGain(); o.type = 'square'; o.frequency.value = 880;
      g.gain.setValueAtTime(.0001, t0 + d); g.gain.exponentialRampToValueAtTime(.25, t0 + d + .02); g.gain.exponentialRampToValueAtTime(.0001, t0 + d + .28);
      o.connect(g); g.connect(ac.destination); o.start(t0 + d); o.stop(t0 + d + .3); }); } catch(e) {}
}
function timerFor_lek(l){
  const s = timerFor(l); if (!s) return '';
  if (TM.lek !== l.id){ timerStopp(); TM.lek = l.id; TM.std = TM.sek = TM.kvar = s; }
  return timerHtml();
}
document.addEventListener('visibilitychange', () => { if (document.visibilityState === 'visible' && TM.int) timerVaken(); });

document.addEventListener('click', e => {
  const tu = e.target.closest('[data-tur]');
  if (tu){ if (tu.dataset.tur === 'slumpa') slumpaTur();
    else { flyttaTur(1); if (location.hash.slice(1) === 'ganget') gangSida(); else { const el = document.getElementById('turnamn'); if (el){ el.textContent = turNamn(); el.classList.remove('pop'); void el.offsetWidth; el.classList.add('pop'); } } }
    return; }
  const gt = e.target.closest('[data-gtabort]');
  if (gt){ const i = +gt.dataset.gtabort; GANG.namn.splice(i, 1); if (GANG.tur > i) GANG.tur--; if (GANG.tur >= gangN()) GANG.tur = 0; sparaGang(); gangSida(); return; }
  const ga = e.target.closest('[data-gang]');
  if (ga){ if (ga.dataset.gang === 'blanda'){ GANG.namn = blanda(GANG.namn); GANG.tur = 0; sparaGang(); gangSida(); }
    else if (ga.dataset.gang === 'rensa'){ if (ga.dataset.saker){ GANG.namn = []; GANG.tur = 0; sparaGang(); gangSida(); } else { ga.dataset.saker = '1'; ga.textContent = T('Säker? Tryck igen'); } }
    return; }
  const kb = e.target.closest('[data-kv]');
  if (kb){ const id = kb.dataset.kv; KV.includes(id) ? (KV = KV.filter(x => x !== id)) : KV.push(id); sparaKV();
    const pa = KV.includes(id); kb.classList.toggle('pa', pa); kb.setAttribute('aria-pressed', pa); kb.textContent = pa ? T('✓ Kvällen') : T('+ Kvällen'); return; }
  const kf = e.target.closest('[data-kvflytt]');
  if (kf){ const i = +kf.dataset.kvflytt, j = i + (+kf.dataset.r); if (j >= 0 && j < KV.length){ [KV[i], KV[j]] = [KV[j], KV[i]]; sparaKV(); kvallSida(); } return; }
  const kbort = e.target.closest('[data-kvbort]');
  if (kbort){ KV = KV.filter(x => x !== kbort.dataset.kvbort); sparaKV(); kvallSida(); return; }
  const ks = e.target.closest('[data-kvslumpa]');
  if (ks){ const kvar = blanda(LEKAR.filter(l => !KV.includes(l.id))); KV = KV.concat(kvar.slice(0, +ks.dataset.kvslumpa).map(l => l.id)); sparaKV(); kvallSida(); return; }
  const kr = e.target.closest('[data-kvrensa]');
  if (kr){ if (kr.dataset.saker){ KV = []; sparaKV(); kvallSida(); } else { kr.dataset.saker = '1'; kr.textContent = T('Säker? Tryck igen'); } return; }
  const ts = e.target.closest('[data-tsek]');
  if (ts){ timerStopp(); TM.sek = TM.kvar = +ts.dataset.tsek; ritaTimer(); return; }
  const tb = e.target.closest('[data-timer]');
  if (tb){ if (tb.dataset.timer === 'start'){ if (TM.int){ timerStopp(); ritaTimer(); } else timerStart(); } else { timerStopp(); TM.kvar = TM.sek; ritaTimer(); } return; }
});
VERKTYG.ganget = gangSida; VERKTYG.kvallen = kvallSida;
